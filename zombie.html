Acerca del Proyecto

Este fue mi primer videojuego desarrollado en Unity. Es un plataformas 2D de acción y aventura en el que el jugador controla a un aldeano que lucha contra zombies. A medida que avanza por los niveles, debe recolectar monedas para progresar, mientras busca derrotar al causante del brote y salvar su aldea.

Lo que realicé

En este proyecto fui responsable de la programación completa del gameplay en Unity, incluyendo el Player, la lógica de enemigos, el sistema de ítems, UI y animaciones. Trabajé en un equipo de tres personas, donde me enfoqué principalmente en la programacion, mientras que los otros integrantes se encargaron del apartado artístico. A pesar de esta división de roles, colaboramos activamente en el diseño del juego, aportando ideas y tomando decisiones en conjunto. Además, participé en el diseño de niveles, contribuyendo a la distribución de espacios y a la experiencia de juego.

Sistema de Player - Ataques

Ataque disparo

Ataque que genera proyectiles. Al colisionar con un enemigo, el proyectil le aplica daño. Este ataque permite al jugador defenderse a mayor distancia.

Codificacion - Ataque Disparo


        using System.Collections;
        using System.Collections.Generic;
        using UnityEngine;
        using UnityEngine.SceneManagement; 

        public class Villager : MonoBehaviour
        {
            public int ammo;
            public GameObject bulletPrefab;
            public GameObject bulletPrefab2;
            public BulletManager bulletManager;
            public AudioSource audioSource;
            public AudioClip soundShootPlayer;
            public AudioClip AmmoClip;


            void Update()
            {
                Shoot();
            }
            void Shoot()
            {
                //Apreta la tecla "X" para poder disparar
                if (Input.GetKeyDown(KeyCode.X) && ammo >= 1 && Time.timeScale > 0)
                {
                    GameObject obj = Instantiate(bulletPrefab);
                    obj.transform.position = transform.position;
                    Vector2 dir = obj.GetComponent().direction;
                    dir.x = direction;
                    obj.GetComponent().direction = dir;   
                    ammo--;
                    bulletManager.SetBullet(ammo);
                    audioSource.PlayOneShot(soundShootPlayer);
                } 
               
            }
            private void OnTriggerEnter2D(Collider2D collision)
            {
                //Recoges la municion de tus balas para poder disparar
                if (collision.gameObject.CompareTag("Ammo"))
                {
                    ammo++;
                    bulletManager.SetBullet(ammo);
                    Destroy(collision.gameObject);
                    audioSource.PlayOneShot(AmmoClip);
                }

            }

        }


        using System.Collections;
        using System.Collections.Generic;
        using UnityEngine;

        public class Bullet : MonoBehaviour
        {
            public Rigidbody2D rb2d;
            public Vector2 direction;
            public int speed;
            public float timer;
            public float MaxTimer;

            void Start()
            {
                rb2d = GetComponent();
            }

            void Update()
            {
                Move();
            }

            void Move()
            {
                //Movimiento y vida de tiempo de la bala
                rb2d.velocity = direction * speed;
                timer += Time.deltaTime;
                if (timer >= MaxTimer)
                {
                    Destroy(gameObject);
                    timer = 0;
                }
            }
        }

    using System.Collections;
    using System.Collections.Generic;
    using UnityEngine;
    using UnityEngine.SceneManagement;
    using Pathfinding;
    using System.IO;
    using Unity.VisualScripting;
    using UnityEngine.UI;

    public class Enemy1 : MonoBehaviour
    {
        void Update()
        {
            //Detienes las acciones del enemigo1 cuando activas el item paralizador
            bool isStopped = player.GetComponent().timerStop.enabled;
            if (isStopped)
            {
                animator?.SetFloat("runEnemyBas", 0);
                if (path != null) path.canMove = false; //Detiene el movimiento del enemigo
            }
            else
            {
                Move(); //Movimiento del enemigo
            }
        }     
    }


    using Pathfinding;
    using System.Collections;
    using System.Collections.Generic;
    using Unity.VisualScripting;
    using UnityEngine;
    using UnityEngine.SceneManagement;
    using UnityEngine.UI;

    public class Enemy2 : MonoBehaviour
    {
           void Update()
           {
                //Detienes las acciones del enemigo2 cuando activas el item paralizador
                bool isStopped = player.GetComponent().timerStop.enabled;

                if (isStopped)
                {
                   //Cuando se activa el item del paralizador
                   StopShootEnemy();//Detiene el disparo del enemigo
                   if (path != null) path.canMove = false; //Detiene el movimiento del enemigo
                   animator?.SetFloat("runEnemyBas", 0);
                }
                else
                {
                   //Cuando no esta activa el item del paralizador
                   Move(); //Movimiento del enemigo
                   StartShootEnemy(); //Disparo del enemigo
                }
           }
           
    }
