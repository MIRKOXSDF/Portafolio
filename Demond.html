Acerca del Proyecto

Proyecto 2D Top Down desarrollado en Unity, donde controlas a un ninja atrapado en un castillo lleno de demonios. Tu objetivo es derrotarlos a todos y encontrar la salida para escapar del castillo en un tiempo determinado.

Lo que realicé

En este proyecto fui responsable de la programación completa del gameplay en Unity, incluyendo el Player, la lógica de enemigos, el sistema de ítems, UI y animaciones. Trabajé en un equipo de tres personas, donde me enfoqué principalmente en la programacion, mientras que los otros integrantes se encargaron del apartado artístico. A pesar de esta división de roles, colaboramos activamente en el diseño del juego, aportando ideas y tomando decisiones en conjunto. Además, participé en el diseño de niveles, contribuyendo a la distribución de espacios y a la experiencia de juego.

Sistema de Player

  • Desarrollo del Player Controller con movimiento en 8 direcciones.

  • Implementación de sistema de disparo direccional (shuriken)

Sistema de Enemigos

  • Programacion de un rango de deteccion para los enemigos.

  • Implementación de persecución mediante Pathfinding.

  • Comportamientos diferenciados por tipo de enemigo.

  • Enemigo volador→ataque a distancia.

  • Enemigo bomba→Al colisionar explota.

Sistema de items

  • Desarrollo del Item Escudo→protección contra daño.

  • Desarrollo del Item de Paralización→detiene acciones enemigas.

Codificacion - Botella de Escudo


    using System;
    using UnityEngine;
    using UnityEngine.SceneManagement;
    using UnityEngine.UI;

    public class Ninja : MonoBehaviour
    {
         public float bottle;
         public float velocityPlayer;
         public GameObject shieldActive;
         public bool shield;
         public SliderShieldUI shieldBar;
         public AudioSource audioSource;
         public float lifePlayer;
         public SliderPlayerLife sliderPlayerLife;
         public bool isDeadPlayer;
         public Animator animator;

         void Update()
         {
            Shield();
         }
        
         void Shield()
         {
           //Activas el escudo con espacio apretando la tecla "ESPACIO"
           if (Input.GetKey(KeyCode.Space) && bottle > 0 )
           {         
              shieldActive.SetActive(true);                              
              shield = true;
              bottle -= 50 * Time.deltaTime / 13f;
              shieldBar.GetComponent().UpdateSliderShield(bottle);        
           }
           else if(Input.GetKeyUp(KeyCode.Space))
           {
              shieldActive.SetActive(false);
              shield = false;
           }
         }
         private void OnTriggerEnter2D(Collider2D collision)
         {
           //Recoges el item escudo
           if (collision.CompareTag("item"))
           {        
              bottle += 50;

              audioSource.PlayOneShot(soundShieldItem);

              shieldBar?.UpdateSliderShield(bottle);

              Destroy(collision.gameObject);
           }
           if (collision.gameObject.CompareTag("Enemybullet"))
           {
              //Cuando eres atacado por los disparos de enemigo 2 , tu escudo los detiene o no
              if (shield) return;

              lifePlayer -= 5;
              sliderPlayerLife.GetComponent().UpdateLifeSlider(lifePlayer);
              if (lifePlayer <= 0)
              {
                 isDeadPlayer = true;
                 animator?.SetBool("muerteplayer", isDeadPlayer);
                 velocityPlayer = 0;
                 Invoke("Dead", 2);
              }
         }    
         void Dead()
         {  //El jugador desaparece "PERDISTES"
            SceneManager.LoadScene("Perdistes");
            Destroy(gameObject);
         }
         private void OnTriggerStay2D(Collider2D collision)
         {
            //Cuando eres atacado por los enemigos 1,2,3 , tu escudo los detiene o no
            if (shield) return;

            float damagePerSecond = 0f;

            switch (collision.tag)
            {
              case "enemy":
                damagePerSecond = 4f;
                break;

              case "enemy2":
                damagePerSecond = 5f;
                break;

              case "enemy3":
                damagePerSecond = 7f; 
                break;
            }
            if (damagePerSecond > 0)
            {
               lifePlayer -= Time.deltaTime * damagePerSecond;
               sliderPlayerLife.GetComponent().UpdateLifeSlider(lifePlayer);

               if (lifePlayer <= 0)
               {
                 animator?.SetBool("muerteplayer", isDeadPlayer);
                 velocityPlayer = 0; 
                 Invoke("Dead", 2); 
               }
            }
         }
    }

Codificacion - Botella de Paralizacion


    using System;
    using UnityEngine;
    using UnityEngine.SceneManagement;
    using UnityEngine.UI;

    public class Ninja : MonoBehaviour
    {
        private bool activeParalyzer;
        private GameObject textMessageItem;
        public Image timerStop;
        public AudioSource audioSource;
        public Animator animator;
        public AIPath path;
        [SerializeField] private float timerPower;
        [SerializeField] private float DelayTimer;
        [SerializeField] private float MaxtimerPower=5;

        void Start()
        {
           timerPower = MaxtimerPower;
        }
        void Update()
        {
            //Activas para utilizar el item paralizador , apretando la tecla "H"
            if (Input.GetKeyDown(KeyCode.H) && activeParalyzer)
            {
               textMessageItem.SetActive(false);
               timerStop.enabled = true;
               InvokeRepeating(nameof(TimerBar), 0f, 0.02f);
            }         
        }
        public void TimerBar()
        {
            timerPower -= Time.deltaTime*DelayTimer;
            timerStop.fillAmount = timerPower / MaxtimerPower;
            if (timerPower <= 0)
            {
               timerPower = 0;
               timerStop.enabled = false;
               CancelInvoke(nameof(TimerBar));
            }
        }     
        private void OnTriggerEnter2D(Collider2D collision)
        {
            //Recoges el item paralizador
            if (collision.gameObject.CompareTag("item2")&&!activeParalyzer)
            {
               audioSource.PlayOneShot(soundShieldItem);
               textMessageItem.SetActive(true);
               activeParalyzer = true;
               Destroy(collision.gameObject);
            }
        }   
    } 





    using System.Collections;
    using System.Collections.Generic;
    using UnityEngine;
    using UnityEngine.SceneManagement;
    using Pathfinding;
    using System.IO;
    using Unity.VisualScripting;
    using UnityEngine.UI;

    public class Enemy1 : MonoBehaviour
    {
        void Update()
        {
            //Detienes las acciones del enemigo1 cuando activas el item paralizador
            bool isStopped = player.GetComponent().timerStop.enabled;
            if (isStopped)
            {
                animator?.SetFloat("runEnemyBas", 0);
                if (path != null) path.canMove = false; //Detiene el movimiento del enemigo
            }
            else
            {
                Move(); //Movimiento del enemigo
            }
        }     
    }


    using Pathfinding;
    using System.Collections;
    using System.Collections.Generic;
    using Unity.VisualScripting;
    using UnityEngine;
    using UnityEngine.SceneManagement;
    using UnityEngine.UI;

    public class Enemy2 : MonoBehaviour
    {
           void Update()
           {
                //Detienes las acciones del enemigo2 cuando activas el item paralizador
                bool isStopped = player.GetComponent().timerStop.enabled;

                if (isStopped)
                {
                   //Cuando se activa el item del paralizador
                   StopShootEnemy();//Detiene el disparo del enemigo
                   if (path != null) path.canMove = false; //Detiene el movimiento del enemigo
                   animator?.SetFloat("runEnemyBas", 0);
                }
                else
                {
                   //Cuando no esta activa el item del paralizador
                   Move(); //Movimiento del enemigo
                   StartShootEnemy(); //Disparo del enemigo
                }
           }
           
    }
