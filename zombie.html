Acerca del Proyecto

Fue el primer Videojuego de plataformas 2D desarrollado en Unity. El jugador controla a un aldeano que debe enfrentarse a zombies mientras avanza por distintos niveles, evitando obstáculos, recolectando monedas y buscando el portal para completar el nivel.

Ataque Disparo

Se desarrolló un sistema de disparo que genera proyectiles. Al colisionar con un enemigo, el proyectil detecta el impacto y aplica daño, permitiendo atacar a distancia.

Ataque Meele

Se desarrolló un sistema de ataque meele basado en la lógica del disparo. En vez de generar proyectiles, se instancia un trinche por un corto tiempo, creando la sensación de un golpe directo a los enemigos cercanos.

        using System.Collections;
        using System.Collections.Generic;
        using UnityEngine;
        using UnityEngine.SceneManagement; 

        public class Villager : MonoBehaviour
        {
            public int ammo;
            public GameObject bulletPrefab;
            public GameObject bulletPrefab2;
            public BulletManager bulletManager;
            public AudioSource audioSource;
            public AudioClip soundShootPlayer;
            public AudioClip AmmoClip;


            void Update()
            {
                Shoot();
            }
            void Shoot()
            {
                //Apreta la tecla "X" para poder disparar
                if (Input.GetKeyDown(KeyCode.X) && ammo >= 1 && Time.timeScale > 0)
                {
                    GameObject obj = Instantiate(bulletPrefab);
                    obj.transform.position = transform.position;
                    Vector2 dir = obj.GetComponent().direction;
                    dir.x = direction;
                    obj.GetComponent().direction = dir;   
                    ammo--;
                    bulletManager.SetBullet(ammo);
                    audioSource.PlayOneShot(soundShootPlayer);
                } 
               
            }
            private void OnTriggerEnter2D(Collider2D collision)
            {
                //Recoges la municion de tus balas para poder disparar
                if (collision.gameObject.CompareTag("Ammo"))
                {
                    ammo++;
                    bulletManager.SetBullet(ammo);
                    Destroy(collision.gameObject);
                    audioSource.PlayOneShot(AmmoClip);
                }

            }

        }


        using System.Collections;
        using System.Collections.Generic;
        using UnityEngine;

        public class Bullet : MonoBehaviour
        {
            public Rigidbody2D rb2d;
            public Vector2 direction;
            public int speed;
            public float timer;
            public float MaxTimer;

            void Start()
            {
                rb2d = GetComponent();
            }

            void Update()
            {
                Move();
            }

            void Move()
            {
                //Movimiento y vida de tiempo de la bala
                rb2d.velocity = direction * speed;
                timer += Time.deltaTime;
                if (timer >= MaxTimer)
                {
                    Destroy(gameObject);
                    timer = 0;
                }
            }
        }
